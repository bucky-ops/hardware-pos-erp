   112→   112→  /* ---- Store hooks ---- */
   113→   113→  const {
   114→   114→    items,
   115→   115→    customerId,
   116→   116→    discountAmount,

   122→   122→    setDiscountAmount,
   123→   123→    clearCart,
   124→   124→    getSubtotal,
   125→   125→  } = usePOSCartStore();
   126→   126→
   127→   127→  const {

   131→   131→    address,
   132→   132→    phone,

   142→   142→  const [loading, setLoading] = useState(true);
   143→   143→  const [searchQuery, setSearchQuery] = useState('');
   144→   144→  const [activeCategoryId, setActiveCategoryId] = useState<string>('all');
   145→   145→  const [paymentMethod, setPaymentMethod] = useState<PaymentMethod>('cash');
   146→   146→  const [amountTendered, setAmountTendered] = useState('');
   147→   147→  const [processing, setProcessing] = useState(false);
   148→   148→  const [showReceipt, setShowReceipt] = useState(false);
   149→   149→  const [receiptData, setReceiptData] = useState<SaleResult | null>(null);
   150→   150→  const [cartOpen, setCartOpen] = useState(false);
   151→   151→
   152→   152→  /* ---- Refs ---- */
   153→   153→  const searchRef = useRef<HTMLInputElement>(null);
   154→   154→
   155→   155→  /* ---- Load settings ---- */
   156→   156→  useEffect(() => {
   157→   157→    loadSettings();

   160→   160→  /* ---- Fetch products, categories, customers ---- */

   170→   170→        const data = await prodRes.json();

   173→   173→      if (catRes.ok) {
   174→   174→        const data = await catRes.json();
   175→   175→        setCategories(Array.isArray(data.categories) ? data.categories : Array.isArray(data.data) ? data.data : []);
   176→   176→      }
   177→   177→      if (custRes.ok) {
   178→   178→        const data = await custRes.json();
   179→   179→        setCustomers(Array.isArray(data.customers) ? data.customers : Array.isArray(data.data) ? data.data : []);
   180→   180→      }
   181→   181→    } catch {
   182→   182→      toast.error('Failed to load data');
   183→   183→    } finally {
   184→   184→      setLoading(false);
   185→   185→    }
   186→   186→  }, []);
   187→   187→
   188→   188→  useEffect(() => {
   189→   189→    fetchData();
   190→   190→  }, [fetchData]);
   191→   191→
   192→   192→  /* ---- Keyboard shortcuts ---- */
   193→   193→  useEffect(() => {
   194→   194→    function handleKeyDown(e: KeyboardEvent) {
   195→   195→      if (e.key === 'F2') {
   196→   196→        e.preventDefault();
   197→   197→        searchRef.current?.focus();
   198→   198→      }
   199→   199→      if (e.key === 'Escape' && document.activeElement === searchRef.current) {
   200→   200→        setSearchQuery('');
   201→   201→        searchRef.current?.blur();
   202→   202→      }
   203→   203→    }
   204→   204→    window.addEventListener('keydown', handleKeyDown);
   205→   205→    return () => window.removeEventListener('keydown', handleKeyDown);
   206→   206→  }, []);
   207→   207→
   208→   208→  /* ---- Computed values ---- */
   209→   209→  const subtotal = getSubtotal();
   210→   210→
   211→   211→  const afterDiscount = Math.max(0, subtotal - discountAmount);
   212→   212→
   213→   213→  const tax = (afterDiscount * taxRate) / 100;
   214→   214→
   215→   215→  const total = afterDiscount + tax;
   216→   216→
   217→   217→  const change = paymentMethod === 'cash'
   218→   218→    ? Math.max(0, parseFloat(amountTendered || '0') - total)
   219→   219→    : 0;
   220→   220→
   221→   221→  const canPay =
   222→   222→    items.length > 0 &&
   223→   223→    total > 0 &&
   224→   224→    (paymentMethod !== 'cash' || parseFloat(amountTendered || '0') >= total);
   225→   225→
   226→   226→  /* ---- Filtered products ---- */
   227→   227→  const filteredProducts = useMemo(() => {
   228→   228→    let result = products;
   229→   229→
   230→   230→    if (activeCategoryId !== 'all') {
   231→   231→      result = result.filter((p) => p.categoryId === activeCategoryId);
   232→   232→    }
   233→   233→
   234→   234→    if (searchQuery.trim()) {
   235→   235→      const q = searchQuery.toLowerCase().trim();
   236→   236→      result = result.filter(
   237→   237→        (p) =>
   238→   238→          p.name.toLowerCase().includes(q) ||
   239→   239→          p.sku.toLowerCase().includes(q) ||
   240→   240→          (p.barcode && p.barcode.toLowerCase().includes(q)),
   241→   241→      );
   242→   242→    }
   243→   243→
   244→   244→    return result;
   245→   245→  }, [products, activeCategoryId, searchQuery]);
   246→   246→
   247→   247→  /* ---- Customer lookup ---- */
   248→   248→  const selectedCustomer = useMemo(
   249→   249→    () => customers.find((c) => c.id === customerId),
   250→   250→    [customers, customerId],
   251→   251→  );
   252→   252→
   253→   253→  /* ---- Add product to cart ---- */
   254→   254→  const handleAddToCart = useCallback(
   255→   255→    (product: Product) => {
   256→   256→      if (product.currentStock <= 0) {
   257→   257→        toast.error(`${product.name} is out of stock`);
   258→   258→        return;
   259→   259→      }
   260→   260→
   261→   261→      const existing = items.find((i) => i.productId === product.id);
   262→   262→      if (existing && existing.quantity >= product.currentStock) {
   263→   263→        toast.error(`Maximum stock reached for ${product.name}`);
   264→   264→        return;
   265→   265→      }
   266→   266→
   267→   267→      addItem({
   268→   268→        productId: product.id,
   269→   269→        productName: product.name,
   270→   270→        sku: product.sku,
   271→   271→        quantity: 1,
   272→   272→        unitPrice: product.sellingPrice,
   273→   273→        costPrice: product.costPrice,
   274→   274→        discount: 0,
   275→   275→        maxDiscount: product.maxDiscount,
   276→   276→      });
   277→   277→      toast.success(`Added ${product.name}`, { description: '1 unit added to cart' });
   278→   278→    },
   279→   279→    [items, addItem],
   280→   280→  );
   281→   281→
   282→   282→  /* ---- Process sale ---- */
   283→   283→  const handleProcessPayment = useCallback(async () => {
   284→   284→    if (items.length === 0 || processing) return;
   285→   285→
   286→   286→    setProcessing(true);
   287→   287→
   288→   288→    try {
   289→   289→      /* -- Step 1: Pre-flight validation -- */
   290→   290→      const ids = items.map((i) => i.productId).join(',');
   291→   291→      const validateRes = await fetch(`/api/products?ids=${ids}&isActive=true`);
   292→   292→
   293→   293→      if (validateRes.ok) {
   294→   294→        const vData = await validateRes.json();
   295→   295→        const validProducts: Product[] = vData.products ?? vData;
   296→   296→        const validIds = new Set(validProducts.map((p: Product) => p.id));
   297→   297→
   298→   298→        const invalidItems = items.filter((i) => !validIds.has(i.productId));
   299→   299→        if (invalidItems.length > 0) {
   300→   300→          for (const inv of invalidItems) {
   301→   301→            removeItem(inv.productId);
   302→   302→            toast.error(`Removed: ${inv.productName}`, {
   303→   303→              description: 'Product no longer available or inactive',
   304→   304→            });
   305→   305→          }
   306→   306→          if (items.length - invalidItems.length === 0) {
   307→   307→            setProcessing(false);
   308→   308→            return;
   309→   309→          }
   310→   310→        }
   311→   311→
   312→   312→        // Check stock for remaining items
   313→   313→        const validItems = items.filter((i) => validIds.has(i.productId));
   314→   314→        const stockMap = new Map(validProducts.map((p: Product) => [p.id, p.currentStock]));
   315→   315→        const outOfStock = validItems.filter(
   316→   316→          (i) => (stockMap.get(i.productId) ?? 0) < i.quantity,
   317→   317→        );
   318→   318→
   319→   319→        if (outOfStock.length > 0) {
   320→   320→          for (const oos of outOfStock) {
   321→   321→            removeItem(oos.productId);
   322→   322→            const currentStock = stockMap.get(oos.productId) ?? 0;
   323→   323→            toast.error(`Removed: ${oos.productName}`, {
   324→   324→              description: currentStock > 0
   325→   325→                ? `Only ${currentStock} in stock (need ${oos.quantity})`
   326→   326→                : 'Out of stock',
   327→   327→            });
   328→   328→          }
   329→   329→          if (validItems.length - outOfStock.length === 0) {
   330→   330→            setProcessing(false);
   331→   331→            return;
   332→   332→          }
   333→   333→        }
   334→   334→      }
   335→   335→
   336→   336→      /* -- Step 2: Submit sale -- */
   337→   337→      const salePayload = {
   338→   338→        customerId: customerId || undefined,
   339→   339→        paymentMethod,
   340→   340→        amountTendered: paymentMethod === 'cash' ? parseFloat(amountTendered || '0') : undefined,
   341→   341→        discountAmount,
   342→   342→        items: items.map((i) => ({
   343→   343→          productId: i.productId,
   344→   344→          quantity: i.quantity,
   345→   345→          unitPrice: i.unitPrice,
   346→   346→          costPrice: i.costPrice,
   347→   347→          discount: i.discount,
   348→   348→        })),
   349→   349→      };
   350→   350→
   351→   351→      const saleRes = await fetch('/api/sales', {
   352→   352→        method: 'POST',
   353→   353→        headers: { 'Content-Type': 'application/json' },
   354→   354→        body: JSON.stringify(salePayload),
   355→   355→      });
   356→   356→
   357→   357→      if (!saleRes.ok) {
   358→   358→        const errData = await saleRes.json().catch(() => ({}));
   359→   359→
   360→   360→        if (saleRes.status === 422 && errData.invalidItems) {
   361→   361→          const invalids: InvalidItem[] = errData.invalidItems;
   362→   362→          for (const inv of invalids) {
   363→   363→            removeItem(inv.productId);
   364→   364→            toast.error(`Removed: ${inv.productName || 'Item'}`, {
   365→   365→              description: inv.reason || 'Validation failed',
   366→   366→            });
   367→   367→          }
   368→   368→        } else {
   369→   369→          toast.error(errData.error || 'Sale failed', {
   370→   370→            description: 'Please try again',
   371→   371→          });
   372→   372→        }
   373→   373→        setProcessing(false);
   374→   374→        return;
   375→   375→      }
   376→   376→
   377→   377→      /* -- Step 3: Success -- */
   378→   378→      const saleResult: SaleResult = await saleRes.json();
   379→   379→
   380→   380→      setReceiptData(saleResult);
   381→   381→      setShowReceipt(true);
   382→   382→      clearCart();
   383→   383→      setAmountTendered('');
   384→   384→      setDiscountAmount(0);
   385→   385→      setCustomer(null);
   386→   386→      setPaymentMethod('cash');
   387→   387→      fetchData();
   388→   388→      toast.success('Sale completed successfully!');
   389→   389→    } catch {
   390→   390→      toast.error('Network error', { description: 'Please check your connection' });
   391→   391→    } finally {
   392→   392→      setProcessing(false);
   393→   393→    }
   394→   394→  }, [
   395→   395→    items, processing, customerId, paymentMethod, amountTendered, discountAmount,
   396→   396→    removeItem, clearCart, setDiscountAmount, setCustomer, fetchData,
   397→   397→  ]);
   398→   398→
   399→   399→  /* ---- Barcode handler (search triggers add if exact barcode match) ---- */

   401→   401→    (e: React.KeyboardEvent<HTMLInputElement>) => {

   403→   403→        const q = searchQuery.trim().toLowerCase();
   404→   404→        const match = products.find(
   405→   405→          (p) =>
   406→   406→            (p.barcode && p.barcode.toLowerCase() === q) ||
   407→   407→            p.sku.toLowerCase() === q,
   408→   408→        );
   409→   409→        if (match) {
   410→   410→          handleAddToCart(match);
   411→   411→          setSearchQuery('');
   412→   412→        }
   413→   413→      }
   414→   414→    },
   415→   415→    [searchQuery, products, handleAddToCart],
   416→   416→  );
   417→   417→
   418→   418→  /* ================================================================
   419→   419→     RENDER — Left Panel: Product Grid
   420→   420→     ================================================================ */
   421→   421→  const renderProductGrid = () => {
   422→   422→    if (loading) {
   423→   423→      return (
   424→   424→        <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-3 xl:grid-cols-4 gap-3 p-4">
   425→   425→          {Array.from({ length: 12 }).map((_, i) => (
   426→   426→            <div key={i} className="space-y-2">
   427→   427→              <Skeleton className="h-24 w-full rounded-lg" />
   428→   428→              <Skeleton className="h-4 w-3/4" />
   429→   429→              <Skeleton className="h-4 w-1/2" />
   430→   430→            </div>
   431→   431→          ))}
   432→   432→        </div>
   433→   433→      );
   434→   434→    }
   435→   435→
   436→   436→    return (
   437→   437→      <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-3 xl:grid-cols-4 gap-3 p-4">
   438→   438→        {filteredProducts.length === 0 ? (
   439→   439→          <div className="col-span-full flex flex-col items-center justify-center py-16 text-muted-foreground">
   440→   440→            <Package className="size-12 mb-3 opacity-40" />
   441→   441→            <p className="text-sm font-medium">No products found</p>
   442→   442→            <p className="text-xs mt-1">Try adjusting your search or category filter</p>
   443→   443→          </div>
   444→   444→        ) : (
   445→   445→          filteredProducts.map((product) => (
   446→   446→            <button
   447→   447→              key={product.id}
   448→   448→              onClick={() => handleAddToCart(product)}
   449→   449→              className={cn(
   450→   450→                'relative flex flex-col items-start gap-1.5 rounded-lg border p-3 text-left transition-all hover:shadow-md hover:border-emerald-400 dark:hover:border-emerald-600 active:scale-[0.98] cursor-pointer bg-card',
   451→   451→                product.currentStock <= 0 && 'opacity-50 cursor-not-allowed hover:shadow-none hover:border-border',
   452→   452→              )}
   453→   453→              disabled={product.currentStock <= 0}
   454→   454→            >
   455→   455→              {/* Stock badge */}
   456→   456→              <Badge
   457→   457→                className={cn(
   458→   458→                  'absolute top-2 right-2 text-[10px] px-1.5 py-0',
   459→   459→                  product.currentStock <= 0
   460→   460→                    ? 'bg-red-100 text-red-700 dark:bg-red-900/40 dark:text-red-400 border-red-200 dark:border-red-800'
   461→   461→                    : product.currentStock <= (product.reorderLevel ?? 5)
   462→   462→                      ? 'bg-amber-100 text-amber-700 dark:bg-amber-900/40 dark:text-amber-400 border-amber-200 dark:border-amber-800'
   463→   463→                      : 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/40 dark:text-emerald-400 border-emerald-200 dark:border-emerald-800',
   464→   464→                )}
   465→   465→                variant="outline"
   466→   466→              >
   467→   467→                {product.currentStock <= 0 ? 'Out' : `${product.currentStock}`}
   468→   468→              </Badge>
   469→   469→
   470→   470→              <p className="text-sm font-medium leading-tight pr-14 line-clamp-2">
   471→   471→                {product.name}
   472→   472→              </p>
   473→   473→              {product.category && (
   474→   474→                <p className="text-[11px] text-muted-foreground line-clamp-1">
   475→   475→                  {product.category.name}
   476→   476→                </p>
   477→   477→              )}
   478→   478→              <p className="text-base font-bold text-emerald-700 dark:text-emerald-400 mt-auto">
   479→   479→                {formatCurrency(product.sellingPrice, currencySymbol)}
   480→   480→              </p>
   481→   481→            </button>
   482→   482→          ))
   483→   483→        )}
   484→   484→      </div>
   485→   485→    );
   486→   486→  };
   487→   487→
   488→   488→  /* ================================================================
   489→   489→     RENDER — Right Panel: Cart
   490→   490→     ================================================================ */
   491→   491→  const renderCartPanel = () => {
   492→   492→    return (
   493→   493→      <div className="flex flex-col h-full bg-card border-l">
   494→   494→        {/* Cart header */}
   495→   495→        <div className="flex items-center justify-between p-4 border-b">
   496→   496→          <div className="flex items-center gap-2">
   497→   497→            <ShoppingCart className="size-5 text-emerald-600 dark:text-emerald-400" />

  1140→  1140→   Utility: Quick tender amounts (rounded-up bills)

  1142→  1142→function getQuickTenderAmounts(total: number): number[] {
  1143→  1143→  if (total <= 0) return [];
  1144→  1144→  const amounts: number[] = [];
  1145→  1145→  const rounded = Math.ceil(total);
  1146→  1146→  const denominations = [5, 10, 20, 50, 100, 200, 500];
  1147→  1147→  for (const d of denominations) {
  1148→  1148→    const amt = Math.ceil(total / d) * d;
  1149→  1149→    if (amt > total && !amounts.includes(amt)) {
  1150→  1150→      amounts.push(amt);
  1151→  1151→    }
  1152→  1152→    if (amounts.length >= 3) break;
  1153→  1153→  }
  1154→  1154→  // Add the exact total rounded up
  1155→  1155→  if (!amounts.includes(rounded) && rounded > total) {
  1156→  1156→    amounts.unshift(rounded);
  1157→  1157→  }
  1158→  1158→  return amounts.slice(0, 4);
  1159→  1159→}
